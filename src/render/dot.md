# Render dot

The layout is DanKG's own, not graphviz's: node positions come out as
pinned `pos` attributes, so `neato -n -Tsvg` reproduces exactly what
`crate::layout` (not yet converted) computed. Plain `dot -Tsvg` ignores
pinned positions and re-lays the graph out on its own, but the `rank=same`
groups and containment edge weights this renderer also writes mean even
that re-layout still agrees with DanKG about which node belongs on which
layer. Edge geometry itself is deliberately never emitted: graphviz wants
B-spline control points, DanKG only has polylines, and converting between
the two to satisfy a flag most people will never pass is not worth risking
an arithmetic bug over.

```rust name=module_doc path=render/dot.rs
//! points, DanKG has polylines, and converting between them to satisfy a flag
//! most people will not pass is not worth a bug in the arithmetic.

use crate::graph::{EdgeKind, Graph, Node, NodeKind};
use crate::layout::Layout;
use std::fmt::Write as _;

/// Graphviz measures position in points and size in inches.
const PER_INCH: f64 = 72.0;
```

A named code block gets its own tint and monospace font, the same
distinction [`render::mermaid`](mermaid.md) draws with a `classDef` and
`tui/draw.rs` draws with a border glyph; an unresolved node is drawn
dashed and muted, since a dangling link is signal worth noticing, not
something to hide behind a normal-looking box.

```rust name=render path=render/dot.rs
pub fn render(graph: &Graph, layout: &Layout) -> String {
    let mut out = String::new();
    out.push_str(
        "// Generated by dankg.\n\
         // `neato -n -Tsvg` draws DanKG's layout exactly; `dot -Tsvg` re-lays it out.\n",
    );
    out.push_str("digraph dankg {\n");
    out.push_str("  graph [rankdir=TB, splines=polyline, bgcolor=\"transparent\"];\n");
    out.push_str(
        "  node [shape=box, style=\"rounded,filled\", fillcolor=\"#ffffff\", \
         color=\"#3c3c3c\", fontname=\"Helvetica\", fontsize=11];\n",
    );
    out.push_str("  edge [color=\"#707070\", arrowsize=0.7];\n");

    if layout.nodes.is_empty() {
        out.push_str("}\n");
        return out;
    }
    out.push('\n');

    for laid in &layout.nodes {
        let node = graph.node(&laid.id);
        let _ = write!(
            out,
            "  {} [label={}, pos=\"{},{}!\", width={:.2}, height={:.2}",
            quote(&laid.id.to_string()),
            quote(node.map(|n| n.title.as_str()).unwrap_or(&laid.id.slug)),
            laid.x,
            // Graphviz counts y upwards from the bottom left.
            layout.height - laid.y,
            laid.width as f64 / PER_INCH,
            laid.height as f64 / PER_INCH
        );
        if let Some(node) = node {
            if !node.resolved {
                // Dangling links are signal. A placeholder looks like one.
                out.push_str(", style=\"rounded,dashed\", color=\"#a0a0a0\"");
                out.push_str(", fontcolor=\"#a0a0a0\"");
            } else if node.kind == NodeKind::Block {
                // A named code block reads as code, not prose: monospace,
                // and a tint distinct from an ordinary heading's white.
                out.push_str(", fillcolor=\"#eef2ff\", fontname=\"Menlo\"");
            }
            let _ = write!(out, ", tooltip={}", quote(&location(node)));
        }
        out.push_str("];\n");
    }

    // Layer membership, so graphviz's own engine ranks the graph the way DanKG
    // did even when the positions are thrown away.
    for layer in layout.layers() {
        if layer.len() < 2 {
            continue;
        }
        out.push_str("  { rank=same;");
        for node in layer {
            let _ = write!(out, " {};", quote(&node.id.to_string()));
        }
        out.push_str(" }\n");
    }

    out.push('\n');
    for edge in &layout.edges {
        if !layout.is_drawn(edge) {
            continue;
        }
        let _ = write!(
            out,
            "  {} -> {}",
            quote(&edge.from.to_string()),
            quote(&edge.to.to_string())
        );

        let mut attrs: Vec<String> = Vec::new();
        if edge.reciprocated {
            attrs.push("dir=none".to_string());
        }
        if edge.kind == EdgeKind::Contains {
            // Matches the layout's own weighting: containment is the skeleton.
            attrs.push("weight=2".to_string());
            attrs.push("color=\"#c0c0c0\"".to_string());
        }
        if !attrs.is_empty() {
            let _ = write!(out, " [{}]", attrs.join(", "));
        }
        out.push_str(";\n");
    }

    out.push_str("}\n");
    out
}

fn location(node: &Node) -> String {
    if node.resolved {
        format!("{}:{}", node.file, node.line)
    } else {
        format!("{} (unresolved)", node.id)
    }
}
```

```rust name=quote path=render/dot.rs
/// Always quoted. Node ids carry `#` and `/`, and a heading can carry anything.
fn quote(value: &str) -> String {
    let mut out = String::with_capacity(value.len() + 2);
    out.push('"');
    for c in value.chars() {
        match c {
            '"' => out.push_str("\\\""),
            '\\' => out.push_str("\\\\"),
            '\n' => out.push_str("\\n"),
            c => out.push(c),
        }
    }
    out.push('"');
    out
}
```

## Tests

```rust name=tests path=render/dot.rs
#[cfg(test)]
mod tests {
    use super::*;
    use crate::graph::graph_of;
    use crate::layout::{self, NODE_HEIGHT};

    fn dot(files: &[(&str, &str)]) -> String {
        let graph = graph_of(files);
        let laid = layout::layout(&graph);
        render(&graph, &laid)
    }

    #[test]
    fn an_empty_graph_is_still_a_valid_digraph() {
        let out = render(&Graph::default(), &Layout::default());
        assert!(out.contains("digraph dankg {"));
        assert!(out.trim_end().ends_with('}'));
        assert!(!out.contains("->"));
    }

    #[test]
    fn nodes_carry_their_id_label_and_pinned_position() {
        let out = dot(&[("a.md", "# One\n\n## Two\n")]);
        assert!(out.contains("\"a#one\" [label=\"One\""), "{out}");
        assert!(out.contains("pos=\""), "{out}");
        assert!(out.contains("!\""), "positions are pinned for neato -n");
        assert!(out.contains("tooltip=\"a.md:1\""), "{out}");
    }

    #[test]
    fn the_y_axis_is_flipped_for_graphviz() {
        let graph = graph_of(&[("a.md", "# One\n\n## Two\n")]);
        let laid = layout::layout(&graph);
        let out = render(&graph, &laid);

        let top = laid.nodes.iter().find(|n| n.rank == 0).unwrap();
        let expected = laid.height - top.y;
        assert!(
            out.contains(&format!("pos=\"{},{}!\"", top.x, expected)),
            "the topmost node should have the largest y; {out}"
        );
        assert!(expected > laid.height / 2);
    }

    #[test]
    fn containment_is_weighted_so_graphviz_agrees_about_layers() {
        let out = dot(&[("a.md", "# One\n\n## Two\n")]);
        assert!(out.contains("\"a#one\" -> \"a#two\" [weight=2"), "{out}");
        assert!(out.contains("{ rank=same;") || out.lines().all(|l| !l.contains("rank=same")));
    }

    #[test]
    fn a_reciprocated_pair_becomes_one_undirected_edge() {
        let out = dot(&[
            ("a.md", "# A\n\n[b](b.md#b)\n"),
            ("b.md", "# B\n\n[a](a.md#a)\n"),
        ]);
        assert_eq!(out.matches("dir=none").count(), 1, "one line, not two: {out}");
        assert!(out.contains("\"a#a\" -> \"b#b\" [dir=none]"), "{out}");
    }

    #[test]
    fn a_block_node_is_tinted_and_monospaced() {
        let out = dot(&[("a.md", "# One\n\n```sh name=setup\n:\n```\n")]);
        assert!(out.contains("\"a#setup\" [label=\"setup\""), "{out}");
        assert!(out.contains("fillcolor=\"#eef2ff\""), "{out}");
        assert!(out.contains("fontname=\"Menlo\""), "{out}");
    }

    #[test]
    fn an_unresolved_node_is_dashed_and_muted() {
        let out = dot(&[("a.md", "# One\n\n[gone](nowhere.md#lost)\n")]);
        assert!(out.contains("\"rounded,dashed\""), "{out}");
        assert!(out.contains("(unresolved)"), "{out}");
    }

    #[test]
    fn a_rank_group_is_written_for_every_shared_layer() {
        let out = dot(&[("a.md", "# One\n\n## Two\n\n## Three\n")]);
        let group = out.lines().find(|l| l.contains("rank=same")).expect("a shared layer");
        assert!(group.contains("\"a#two\";"), "{group}");
        assert!(group.contains("\"a#three\";"), "{group}");
    }

    #[test]
    fn quotes_and_backslashes_in_a_heading_survive() {
        assert_eq!(quote(r#"say "hi"\"#), r#""say \"hi\"\\""#);
        let out = dot(&[("a.md", "# A \\\"quoted\\\" heading\n")]);
        assert!(out.contains(r#"label="A \"quoted\" heading""#), "{out}");
    }

    #[test]
    fn box_size_is_reported_in_inches() {
        let out = dot(&[("a.md", "# One\n")]);
        let expected = format!("height={:.2}", NODE_HEIGHT as f64 / PER_INCH);
        assert!(out.contains(&expected), "{out}");
    }
}
```
