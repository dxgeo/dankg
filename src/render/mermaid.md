# Render mermaid

Mermaid runs its own layout engine and refuses coordinates, so what this
renderer hands it is only ordering and direction: nodes emitted rank by
rank, left to right, in the same order `crate::layout` (not yet converted)
already computed, and edges pointing the way DanKG's own cycle-breaking
([`layout::acyclic`](../layout/acyclic.md)) decided -- so a hand-drawn
Mermaid diagram and DanKG's own HTML/dot output always agree about which
edge points where, even around a cycle one of them had to turn around
internally.

```rust name=module_doc path=render/mermaid.rs
//! Mermaid flowchart output.
//!
//! Mermaid runs its own layout and will not take coordinates, so what it takes
//! from DanKG is the ordering: nodes are emitted rank by rank, left to right,
//! which is enough for its layered engine to land on the same shape. Edge
//! direction is DanKG's too, including which edge of a cycle was turned
//! around, so the two drawings agree about what points where.
//!
//! Identifiers are `n0`, `n1`, ... rather than node ids. Mermaid's identifier
//! grammar does not admit `#` or `/`, and quoting them is not portable across
//! its versions.

use crate::graph::{EdgeKind, Graph, NodeId, NodeKind};
use crate::layout::Layout;
use std::fmt::Write as _;
```

A named code block reads as code, not prose, so it gets a distinct tint --
the same distinction [`render::dot`](dot.md) draws with `fillcolor` and
`tui/draw.rs` draws with a different border glyph, each renderer's own
way of drawing the same underlying `NodeKind::Block` fact.

```rust name=render path=render/mermaid.rs
pub fn render(graph: &Graph, layout: &Layout) -> String {
    let mut out = String::new();
    out.push_str("%% Generated by dankg. Node order is DanKG's layout, rank by rank.\n");
    out.push_str("flowchart TB\n");

    if layout.nodes.is_empty() {
        return out;
    }
    out.push_str("    classDef dangling stroke-dasharray:4 3,stroke:#a0a0a0,color:#a0a0a0\n");
    // A named code block reads as code, not prose: a tint distinct from an
    // ordinary heading's default, the same distinction `dot.rs` draws with
    // fillcolor and `tui/draw.rs` draws with a different border glyph.
    out.push_str("    classDef block fill:#eef2ff,stroke:#3c3c3c\n");

    // Emitted in layout order, so the identifiers themselves read top to
    // bottom and left to right.
    let names: Vec<(NodeId, String)> = layout
        .nodes
        .iter()
        .enumerate()
        .map(|(i, n)| (n.id.clone(), format!("n{i}")))
        .collect();
    let name_of = |id: &NodeId| {
        names.iter().find(|(node, _)| node == id).map(|(_, name)| name.as_str())
    };

    let mut dangling: Vec<&str> = Vec::new();
    let mut blocks: Vec<&str> = Vec::new();
    for (id, name) in &names {
        let node = graph.node(id);
        let title = node.map(|n| n.title.as_str()).unwrap_or(&id.slug);
        let _ = writeln!(out, "    {name}[\"{}\"]", escape(title));
        if node.is_some_and(|n| !n.resolved) {
            dangling.push(name);
        } else if node.is_some_and(|n| n.kind == NodeKind::Block) {
            blocks.push(name);
        }
    }

    for edge in &layout.edges {
        if !layout.is_drawn(edge) {
            continue;
        }
        let (Some(from), Some(to)) = (name_of(&edge.from), name_of(&edge.to)) else {
            continue;
        };
        let arrow = match (edge.kind, edge.reciprocated) {
            // Containment is the document skeleton, so it is drawn heavier.
            (EdgeKind::Contains, _) => "==>",
            (EdgeKind::Link, true) => "---",
            (EdgeKind::Link, false) => "-->",
        };
        let _ = writeln!(out, "    {from} {arrow} {to}");
    }

    if !dangling.is_empty() {
        let _ = writeln!(out, "    class {} dangling", dangling.join(","));
    }
    if !blocks.is_empty() {
        let _ = writeln!(out, "    class {} block", blocks.join(","));
    }
    out
}
```

```rust name=escape path=render/mermaid.rs
/// Mermaid reads `#` as the start of an entity code, which is also the only
/// way to escape anything, so every character it could misread becomes one.
fn escape(title: &str) -> String {
    let mut out = String::with_capacity(title.len());
    for c in title.chars() {
        match c {
            '#' => out.push_str("#35;"),
            '"' => out.push_str("#quot;"),
            '<' => out.push_str("#lt;"),
            '>' => out.push_str("#gt;"),
            '\n' => out.push_str("<br/>"),
            c => out.push(c),
        }
    }
    out
}
```

## Tests

```rust name=tests path=render/mermaid.rs
#[cfg(test)]
mod tests {
    use super::*;
    use crate::graph::graph_of;
    use crate::layout;

    fn mermaid(files: &[(&str, &str)]) -> String {
        let graph = graph_of(files);
        let laid = layout::layout(&graph);
        render(&graph, &laid)
    }

    #[test]
    fn an_empty_graph_is_still_a_valid_flowchart() {
        let out = render(&Graph::default(), &Layout::default());
        assert_eq!(out.lines().last(), Some("flowchart TB"));
    }

    #[test]
    fn nodes_are_numbered_in_layout_order() {
        let out = mermaid(&[("a.md", "# One\n\n## Two\n")]);
        assert!(out.contains("    n0[\"One\"]"), "{out}");
        assert!(out.contains("    n1[\"Two\"]"), "{out}");
        assert!(out.find("n0[").unwrap() < out.find("n1[").unwrap());
    }

    #[test]
    fn containment_is_drawn_heavier_than_a_link() {
        let out = mermaid(&[("a.md", "# One\n\n## Two\n")]);
        assert!(out.contains("n0 ==> n1"), "{out}");
    }

    #[test]
    fn a_one_way_link_keeps_its_arrow() {
        let out = mermaid(&[("a.md", "# A\n\n[b](b.md#b)\n"), ("b.md", "# B\n")]);
        assert!(out.contains(" --> "), "{out}");
        assert!(!out.contains(" --- "), "{out}");
    }

    #[test]
    fn a_reciprocated_pair_becomes_one_line_with_no_arrow() {
        let out = mermaid(&[
            ("a.md", "# A\n\n[b](b.md#b)\n"),
            ("b.md", "# B\n\n[a](a.md#a)\n"),
        ]);
        assert_eq!(out.matches(" --- ").count(), 1, "{out}");
        assert!(!out.contains(" --> "), "{out}");
    }

    #[test]
    fn unresolved_nodes_are_collected_into_one_class_line() {
        let out = mermaid(&[(
            "a.md",
            "# One\n\n[gone](nowhere.md#lost) and [also](elsewhere.md#missing)\n",
        )]);
        let class = out.lines().find(|l| l.starts_with("    class ")).expect("a class line");
        assert_eq!(class.matches(',').count(), 1, "both placeholders, one line: {class}");
        assert!(out.contains("classDef dangling"));
    }

    #[test]
    fn a_block_node_gets_its_own_class() {
        let out = mermaid(&[("a.md", "# One\n\n```sh name=setup\n:\n```\n")]);
        assert!(out.contains("classDef block"), "{out}");
        let class = out.lines().find(|l| l.starts_with("    class ") && l.ends_with(" block")).expect("a block class line");
        assert_eq!(class.matches(',').count(), 0, "one block node: {class}");
    }

    #[test]
    fn characters_mermaid_would_misread_are_escaped() {
        assert_eq!(escape("C# <b> \"q\""), "C#35; #lt;b#gt; #quot;q#quot;");
        let out = mermaid(&[("a.md", "# Issue #12\n")]);
        assert!(out.contains("n0[\"Issue #35;12\"]"), "{out}");
    }

    #[test]
    fn a_cycle_keeps_the_direction_the_author_wrote() {
        let out = mermaid(&[
            ("a.md", "# A\n\n[b](b.md#b)\n"),
            ("b.md", "# B\n\n[c](c.md#c)\n"),
            ("c.md", "# C\n\n[a](a.md#a)\n"),
        ]);
        // Three one-way links around a loop: three arrows, none of them undone
        // by the layout having reversed one of them internally.
        assert_eq!(out.matches(" --> ").count(), 3, "{out}");
    }
}
```
